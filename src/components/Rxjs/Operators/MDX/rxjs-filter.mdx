:::info
:diamond_shape_with_a_dot_inside: The filter operator is one of the most commonly used operators in RxJS.
It allows you to selectively emit values from an observable based on some condition.
:::

```ts
import { from } from 'rxjs';
import { filter } from 'rxjs/operators';

const observable$ = from([1, 2, 3, 4, 5, 6]);

const filtered$ = observable$.pipe(
  filter(value => value % 2 !== 0)
);

filtered$.subscribe(
  value => console.log(value),
  err => console.error(err),
  () => console.log('Observable complete')
);

// Output
// 1
// 3
// 5
```
